    %% ================= IDENTITY SERVICE =================
    namespace IdentityService {
        class User {
            +Int id
            +String email
            +String full_name
            +String phone
            +String address
            +Date dob
            +Enum gender
            +Status status
            +register()
            +updateProfile()
        }
        class SocialAccount {
            +Int id
            +String provider
            +String provider_id
        }
        class Role {
            +Int id
            +String name
        }
        class Permission {
            +Int id
            +String code
        }
    }
    User "1" -- "*" SocialAccount : has
    User "*" -- "*" Role : assigned_to
    Role "*" -- "*" Permission : includes

            +Int id

            +Int user_id
            +Status status
            +JSONB price_snapshot
            +JSONB promotion_snapshot
            +Timestamp created_at
            +calculateTotal()
        }
        class Passenger {
            +String full_name
            +Date dob
            +Enum age_group
            +String id_card_number
        }
        class BookingNote {
            +String content
        }
        class Cancellation {
            +String reason
